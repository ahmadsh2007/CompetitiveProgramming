struct EulerLCA {
    int n, timer = 0;
    vector<int> depth, first, tour;
    vector<vector<int>> st;

    EulerLCA(int n, const vector<vector<int>>& adj, int root = 1) : n(n) {
        depth.resize(n + 1);
        first.resize(n + 1);
        tour.reserve(2 * n);
        dfs(root, 0, 0, adj);

        int m = tour.size();
        int k = 32 - __builtin_clz(m);
        st.assign(k, vector<int>(m));
        for (int i = 0; i < m; ++i) st[0][i] = tour[i];

        for (int j = 1; (1 << j) <= m; ++j) {
            for (int i = 0; i + (1 << j) <= m; ++i) {
                int left = st[j - 1][i];
                int right = st[j - 1][i + (1 << (j - 1))];
                st[j][i] = (depth[left] < depth[right]) ? left : right;
            }
        }
    }

    void dfs(int u, int p, int d, const vector<vector<int>>& adj) {
        depth[u] = d;
        first[u] = tour.size();
        tour.push_back(u);
        for (int v : adj[u]) {
            if (v != p) {
                dfs(v, u, d + 1, adj);
                tour.push_back(u);
            }
        }
    }

    int lca(int u, int v) {
        int l = first[u], r = first[v];
        if (l > r) swap(l, r);
        int j = 31 - __builtin_clz(r - l + 1);
        int left = st[j][l];
        int right = st[j][r - (1 << j) + 1];
        return (depth[left] < depth[right]) ? left : right;
    }
};
