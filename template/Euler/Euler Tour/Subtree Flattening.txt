// 1-indexed. timer initialized to 0.
int n, timer = 0;
vector<vector<int>> adj;
vector<int> tin, tout;

void dfs(int u, int p = 0) {
    tin[u] = ++timer;
    for (int v : adj[u]) {
        if (v != p) dfs(v, u);
    }
    tout[u] = timer; // [tin[u], tout[u]] covers subtree(u)
}
// Query/Update:
//   Subtree of u: range [tin[u], tout[u]] in Segment Tree
//   Is u ancestor of v?: tin[u] <= tin[v] && tout[u] >= tout[v]
