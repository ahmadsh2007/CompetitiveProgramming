const int MAXN = 10000005;
int phi[MAXN];
vector<int> primes;
bool is_prime[MAXN];

void sieve_phi(int n) {
    fill(is_prime, is_prime + n + 1, true);
    phi[1] = 1;
    for (int i = 2; i <= n; ++i) {
        if (is_prime[i]) {
            primes.push_back(i);
            phi[i] = i - 1;
        }
        for (int p : primes) {
            if (i * p > n) break;
            is_prime[i * p] = false;
            if (i % p == 0) {
                phi[i * p] = phi[i] * p;
                break;
            } else {
                phi[i * p] = phi[i] * (p - 1);
            }
        }
    }
}
